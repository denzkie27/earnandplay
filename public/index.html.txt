    <script>
        // Frontend: no game logic, only UI and animation
        const SYMBOLS = [
            { icon: 'star',      color: '#f1c40f' },
            { icon: 'filter_7',  color: '#e74c3c' },
            { icon: 'diamond',   color: '#3498db' },
            { icon: 'favorite',  color: '#e74c3c' },
            { icon: 'flash_on',  color: '#f39c12' },
            { icon: 'ac_unit',   color: '#1abc9c' },
            { icon: 'whatshot',  color: '#e67e22' },
            { icon: 'bug_report',color: '#27ae60' },
            { icon: 'pets',      color: '#9b59b6' },
            { icon: 'local_florist', color: '#2ecc71' },
            { icon: 'wb_sunny',  color: '#f1c40f' },
            { icon: 'music_note',color: '#e91e63' }
        ];
        const SCATTER_ICON = 'star';
        const CELL_HEIGHT = 120;
        const VISIBLE_ROWS = 3;
        const REEL_COUNT = 5;
        const STRIP_LENGTH = 30;

        // DOM elements
        const balanceSpan = document.getElementById('balance');
        const betInput = document.getElementById('bet');
        const spinBtn = document.getElementById('spinBtn');
        const messageDiv = document.getElementById('message');
        const freeSpinsMsgDiv = document.getElementById('freeSpinsMsg');
        const reelsContainer = document.getElementById('reelsContainer');

        const reelStrips = [];

        // Initialize reels UI
        function createReels() {
            for (let col = 0; col < REEL_COUNT; col++) {
                const column = document.createElement('div');
                column.className = 'reel-column';
                const strip = document.createElement('div');
                strip.className = 'reel-strip';
                column.appendChild(strip);
                reelsContainer.appendChild(column);
                reelStrips.push(strip);
                // Populate with random symbols initially
                for (let i = 0; i < STRIP_LENGTH; i++) {
                    const cell = document.createElement('div');
                    cell.className = 'reel-cell';
                    const sym = SYMBOLS[Math.floor(Math.random() * SYMBOLS.length)];
                    const iconSpan = document.createElement('span');
                    iconSpan.className = 'material-icons';
                    iconSpan.textContent = sym.icon;
                    iconSpan.style.color = sym.color;
                    cell.appendChild(iconSpan);
                    strip.appendChild(cell);
                }
            }
        }

        // Fetch initial state from server
        async function fetchState() {
            try {
                const res = await fetch('/api/state');
                const data = await res.json();
                balanceSpan.textContent = data.balance;
                betInput.max = data.balance;
                updateFreeSpinsMsg(data.freeSpins);
            } catch (err) {
                console.error('Failed to fetch state', err);
            }
        }

        function updateFreeSpinsMsg(count) {
            freeSpinsMsgDiv.textContent = count > 0 ? `🎁 Free spins remaining: ${count}` : "";
        }

        // Prepare a reel strip with the final three symbols at a random index
        function prepareStrip(strip, finalColumnSymbols) {
            strip.innerHTML = '';
            const idx = Math.floor(Math.random() * (STRIP_LENGTH - VISIBLE_ROWS + 1)) + 1;
            for (let i = 0; i < STRIP_LENGTH; i++) {
                const cell = document.createElement('div');
                cell.className = 'reel-cell';
                let sym;
                if (i >= idx - 1 && i <= idx + 1) {
                    sym = finalColumnSymbols[i - (idx - 1)];
                } else {
                    sym = SYMBOLS[Math.floor(Math.random() * SYMBOLS.length)];
                }
                const iconSpan = document.createElement('span');
                iconSpan.className = 'material-icons';
                iconSpan.textContent = sym.icon;
                iconSpan.style.color = sym.color;
                cell.appendChild(iconSpan);
                strip.appendChild(cell);
            }
            return -(idx - 1) * CELL_HEIGHT;
        }

        // Highlight scatter cells
        function highlightScatters(grid, finalTranslateYs) {
            for (let col = 0; col < REEL_COUNT; col++) {
                const strip = reelStrips[col];
                const cells = strip.children;
                const middleIndex = Math.round(-finalTranslateYs[col] / CELL_HEIGHT) + 1;
                for (let row = 0; row < VISIBLE_ROWS; row++) {
                    const cellIndex = middleIndex + row - 1;
                    if (grid[col][row].icon === SCATTER_ICON && cells[cellIndex]) {
                        cells[cellIndex].classList.add('win');
                        setTimeout(() => cells[cellIndex].classList.remove('win'), 1000);
                    }
                }
            }
        }

        // Spin function
        async function spin(isFreeSpin = false) {
            spinBtn.disabled = true;
            messageDiv.textContent = "";

            const bet = parseInt(betInput.value);
            const requestBody = isFreeSpin ? { isFreeSpin: true } : { bet };

            try {
                const res = await fetch('/api/spin', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(requestBody)
                });
                const data = await res.json();
                if (!res.ok) {
                    messageDiv.textContent = data.error || 'Error';
                    spinBtn.disabled = false;
                    return;
                }

                // Update balance display
                balanceSpan.textContent = data.newBalance;
                betInput.max = data.newBalance;
                updateFreeSpinsMsg(data.freeSpinsRemaining);

                // Prepare reels for animation
                const finalGrid = data.grid;
                const finalTranslateYs = [];
                for (let col = 0; col < REEL_COUNT; col++) {
                    const translateY = prepareStrip(reelStrips[col], finalGrid[col]);
                    finalTranslateYs.push(translateY);
                }

                // Animate: start high, then go to final positions
                reelStrips.forEach((strip, col) => {
                    strip.style.transition = 'none';
                    strip.style.transform = `translateY(-${STRIP_LENGTH * CELL_HEIGHT}px)`;
                    void strip.offsetHeight;
                    strip.style.transition = `transform 1.2s cubic-bezier(0.25, 0.1, 0.15, 1) ${col * 0.1}s`;
                    strip.style.transform = `translateY(${finalTranslateYs[col]}px)`;
                });

                const maxDuration = 1.2 + (REEL_COUNT - 1) * 0.1;
                setTimeout(() => {
                    // Highlight scatters if any
                    if (data.scatters >= 3) {
                        highlightScatters(finalGrid, finalTranslateYs);
                    }
                    // Display message
                    let msg = "";
                    if (data.scatters >= 3) {
                        msg = `✨ ${data.scatters} scatters! You win $${data.winAmount}`;
                        if (data.freeSpinsAwarded > 0) msg += ` + ${data.freeSpinsAwarded} free spins!`;
                    } else {
                        msg = "No scatter win. Try again!";
                    }
                    messageDiv.textContent = msg;

                    spinBtn.disabled = false;

                    // If free spins remain, auto-spin after a delay
                    if (data.freeSpinsRemaining > 0) {
                        setTimeout(() => spin(true), 1500);
                    }
                }, maxDuration * 1000 + 200);

            } catch (err) {
                console.error('Spin failed', err);
                messageDiv.textContent = 'Network error';
                spinBtn.disabled = false;
            }
        }

        // Event listeners
        spinBtn.addEventListener('click', () => spin(false));

        // Initialize
        createReels();
        fetchState();
    </script>
